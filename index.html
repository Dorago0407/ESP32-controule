digitalWrite(FL2, HIGH); ledcWrite(C_FL, spd); digitalWrite(FR1, HIGH); digitalWrite(FR2, LOW); ledcWrite(C_FR, spd); digitalWrite(BL1, HIGH); digitalWrite(BL2, LOW); ledcWrite(C_BL, spd); digitalWrite(BR1, LOW); digitalWrite(BR2, HIGH); ledcWrite(C_BR, spd); } void back(int spd) { digitalWrite(FL1, HIGH); digitalWrite(FL2, LOW); ledcWrite(C_FL, spd); digitalWrite(FR1, LOW); digitalWrite(FR2, HIGH); ledcWrite(C_FR, spd); digitalWrite(BL1, LOW); digitalWrite(BL2, HIGH); ledcWrite(C_BL, spd); digitalWrite(BR1, HIGH); digitalWrite(BR2, LOW); ledcWrite(C_BR, spd); } void right(int spd) { digitalWrite(FL1, LOW); digitalWrite(FL2, HIGH); ledcWrite(C_FL, spd); digitalWrite(FR1, LOW); digitalWrite(FR2, HIGH); ledcWrite(C_FR, spd); digitalWrite(BL1, HIGH); digitalWrite(BL2, LOW); ledcWrite(C_BL, spd); digitalWrite(BR1, HIGH); digitalWrite(BR2, LOW); ledcWrite(C_BR, spd); } void left(int spd) { digitalWrite(FL1, HIGH); digitalWrite(FL2, LOW); ledcWrite(C_FL, spd); digitalWrite(FR1, HIGH); digitalWrite(FR2, LOW); ledcWrite(C_FR, spd); digitalWrite(BL1, LOW); digitalWrite(BL2, HIGH); ledcWrite(C_BL, spd); digitalWrite(BR1, LOW); digitalWrite(BR2, HIGH); ledcWrite(C_BR, spd); } void stopRobot(void) { digitalWrite(FL1, LOW); digitalWrite(FL2, LOW); ledcWrite(C_FL, 0); digitalWrite(FR1, LOW); digitalWrite(FR2, LOW); ledcWrite(C_FR, 0); digitalWrite(BL1, LOW); digitalWrite(BL2, LOW); ledcWrite(C_BL, 0); digitalWrite(BR1, LOW); digitalWrite(BR2, LOW); ledcWrite(C_BR, 0); } // 🆕 スマホへ文字列メッセージを送る関数 (元コードのまま) void sendToPhone(String message) { if (deviceConnected && pTxCharacteristic) { pTxCharacteristic->setValue(message.c_str()); pTxCharacteristic->notify(); } } // コマンド処理 void processCommand(String cmd) { Serial.println("CMD: " + cmd); lastCmdTime = millis(); // 最終受信時刻を更新 // 速度変更処理の拡張 (例: "speed:180") if (cmd.startsWith("speed:")) { speed = cmd.substring(6).toInt(); speed = constrain(speed, 0, 255); sendToPhone("STATUS: SPEEDUPDATED"); return; } if (cmd == "mode_auto") { currentMode = MODE_AUTO; sendToPhone("STATUS: MODE AUTO"); return; } else if (cmd == "mode_manual") { currentMode = MODE_MANUAL; stopRobot(); sendToPhone("STATUS: MODE MANUAL"); return; } if (currentMode == MODE_MANUAL) { if (cmd == "forward") { front(speed); sendToPhone("STATUS: FORWARD"); } else if (cmd == "backward"){ back(speed); sendToPhone("STATUS: BACKWARD"); } else if (cmd == "left") { left(speed); sendToPhone("STATUS: TURN LEFT"); } else if (cmd == "right") { right(speed); sendToPhone("STATUS: TURN RIGHT"); } else { stopRobot(); sendToPhone("STATUS: STOP"); } } } // 自律走行処理 (元コードのまま) void autonomousBehavior() { int v1 = analogRead(S1); int v2 = analogRead(S2); if (v1 < 1000 && v2 < 1000) { front(180); } else if (v1 >= 1000) { right(180); } else if (v2 >= 1000) { left(180); } else { stopRobot(); } } class MyCallbacks: public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pCharacteristic) { std::string rxValue = pCharacteristic->getValue(); if (rxValue.length() > 0) { String cmd = ""; for (size_t i = 0; i < rxValue.length(); i++) { cmd += rxValue[i]; } cmd.trim(); processCommand(cmd); } } }; class MyServerCallbacks: public BLEServerCallbacks { void onConnect(BLEServer* pServer) { deviceConnected = true; Serial.println("Connected!"); }; void onDisconnect(BLEServer* pServer) { deviceConnected = false; currentMode = MODE_MANUAL; stopRobot(); pServer->getAdvertising()->start(); } }; void setup() { Serial.begin(115200); pinMode(FR1, OUTPUT); pinMode(FR2, OUTPUT); pinMode(FL1, OUTPUT); pinMode(FL2, OUTPUT); pinMode(BR1, OUTPUT); pinMode(BR2, OUTPUT); pinMode(BL1, OUTPUT); pinMode(BL2, OUTPUT); pinMode(FL_PWM, OUTPUT); pinMode(FR_PWM, OUTPUT); pinMode(BL_PWM, OUTPUT); pinMode(BR_PWM, OUTPUT); pinMode(S1, INPUT); pinMode(S2, INPUT); pinMode(S3, INPUT); pinMode(S4, INPUT); // 🆕 S3, S4も追加設定 // 元コード通りのPWM設定・接続割り当て ledcSetup(C_FL, 1100, 8); ledcSetup(C_FR, 1200, 8); ledcSetup(C_BL, 1300, 8); ledcSetup(C_BR, 1400, 8); ledcAttachPin(FL_PWM, C_BL); ledcAttachPin(FR_PWM, C_BR); ledcAttachPin(BL_PWM, C_FL); ledcAttachPin(BR_PWM, C_FR); stopRobot(); BLEDevice::init(ROBOT_NAME); BLEServer *pServer = BLEDevice::createServer(); pServer->setCallbacks(new MyServerCallbacks()); BLEService *pService = pServer->createService(SERVICE_UUID); // RX（受信ポータル） BLECharacteristic *pRxCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID_RX, BLECharacteristic::PROPERTY_WRITE ); pRxCharacteristic->setCallbacks(new MyCallbacks()); // 🆕 TX（スマホへの送信ポータル） pTxCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID_TX, BLECharacteristic::PROPERTY_NOTIFY ); pTxCharacteristic->addDescriptor(new BLE2902()); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(true); BLEDevice::startAdvertising(); Serial.println("BLE Robot Ready!"); } void loop() { if (currentMode == MODE_AUTO) { autonomousBehavior(); } // 🆕 安全自動停止（手動モード時、1秒間通信が無ければ即時停止） if (currentMode == MODE_MANUAL && millis() - lastCmdTime > SAFETY_TIMEOUT) { stopRobot(); } // 🆕 4つのセンサー(S1~S4)の値をスマホへ送信 (200ms間隔) if (deviceConnected && (millis() - lastSensorSendTime > 200)) { lastSensorSendTime = millis(); int s1Val = analogRead(S1); int s2Val = analogRead(S2); int s3Val = analogRead(S3); int s4Val = analogRead(S4); String sensorData = "SENS:S1=" + String(s1Val) + ",S2=" + String(s2Val) + ",S3=" + String(s3Val) + ",S4=" + String(s4Val); sendToPhone(sensorData); } delay(10); }
